         stx $fc
         ldy #$00
ploop    lda ($fb),y
         beq dprint
         jsr $ffd2
         inc $fb
         bne ploop
         inc $fc
         bne ploop

loader   = *
         *= $0590
         .offs loader-*
main
         lda #$ef
         cmp $dc01
         bne *-3
         jsr ton
         jsr dly
         jsr lod
         jsr tof
         jsr namer
         ldx #$0f

         sta $0570,x

         ldx #$03

         sta buffer,x
         lda $ac,x
